\section{Contesto e obiettivi}

\begin{frame}{Problemi iniziali da risolvere}
  \begin{center}
  \begin{tcbraster}[raster columns=3,raster equal height,raster column skip=8pt,raster row skip=8pt]
    \begin{cardred}{Storage frammentato}
      Tanti file CSV rallentano la \textbf{ricerca dei cluster liberi}
    \end{cardred}
    \begin{cardred}{Formato inefficiente}
      \textbf{CSV}: elevato spazio occupato e \textbf{recovery} complesso
    \end{cardred}
    \begin{cardred}{Sessioni non gestite}
      Nessuna organizzazione in \textbf{sessioni}
    \end{cardred}
    \begin{cardred}{Stato non visibile}
      Nessuno stato centrale \textbf{interno} né \textbf{LED} esterno
    \end{cardred}
    \begin{cardred}{Dati IMU e Barometro grezzi}
      Tutti i dati vengono inviati, \textbf{anche se privi di significato}
    \end{cardred}
    \begin{cardpetrol}{App incompleta}
      Nessun \textbf{modello 3D} del veicolo
    \end{cardpetrol}
    \begin{cardgold}{Dati ECU assenti}
      Nessun accesso ai dati del \textbf{bus CAN}
    \end{cardgold}
  \end{tcbraster}
  \end{center}
\end{frame}
